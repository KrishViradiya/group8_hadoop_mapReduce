  public static String toString(AbstractCounters<Counter, CounterGroupBase<Counter>> c) {
    StringBuilder sb = new StringBuilder();
    List<String> parts = Lists.newArrayList();
    for (CounterGroupBase<? extends Counter> g: c) {
      parts.add(groupToString(g));
    }
    sb.append(StringUtils.join(parts, GROUP_OPEN + "AND" + GROUP_CLOSE));
    return sb.toString();
  }
